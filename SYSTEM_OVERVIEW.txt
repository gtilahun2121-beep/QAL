🧮 AUTHENTICATION FLOWS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

MEMBER REGISTRATION (4 Steps)
┌────────────────────────────────────────────────────────────────┐
│                                                                │
│  Step 1: Phone Number Input                                   │
│  ├─ Input: International format (e.g., +251911223344)         │
│  ├─ Validation: Format check                                  │
│  └─ Action: Send OTP → Toast: "✅ OTP Sent"                  │
│                                                                │
│  Step 2: OTP Verification                                     │
│  ├─ Input: 6-digit code from SMS                              │
│  ├─ Validation: Length check (6 digits)                       │
│  └─ Action: Verify OTP → Toast: "✅ OTP Verified"            │
│                                                                │
│  Step 3: User Details                                         │
│  ├─ Input: Full Name, 4-digit PIN, Fayda Number             │
│  ├─ Validation: Required fields, PIN 4 digits                │
│  └─ Action: Complete → Toast: "✅ Account Created!"           │
│                                                                │
│  Step 4: Success Screen                                       │
│  ├─ Display: Account details summary                          │
│  ├─ Storage: localStorage (demo) → Database (production)      │
│  └─ Navigation: Home or Dashboard                             │
│                                                                │
└────────────────────────────────────────────────────────────────┘

MEMBER LOGIN (2 Steps)
┌────────────────────────────────────────────────────────────────┐
│                                                                │
│  Step 1: Phone Number Input                                   │
│  ├─ Input: Registered phone number                            │
│  ├─ Check: Does account exist?                                │
│  └─ Result: Toast "✅ Phone Found" → Step 2                   │
│                                                                │
│  Step 2: PIN Verification                                     │
│  ├─ Input: 4-digit security PIN                               │
│  ├─ Verify: Does PIN match?                                   │
│  └─ Result: Toast "🎉 Welcome Back!" → Success               │
│                                                                │
└────────────────────────────────────────────────────────────────┘

ADMIN LOGIN (3 Steps)
┌────────────────────────────────────────────────────────────────┐
│                                                                │
│  Step 1: Email Verification                                   │
│  ├─ Input: Admin email address                                │
│  ├─ Check: Is this admin account?                             │
│  └─ Result: Toast "Email Found" → Step 2                      │
│                                                                │
│  Step 2: Master Password                                      │
│  ├─ Input: Admin master password (min 8 chars)               │
│  ├─ Verify: Password hash match?                              │
│  └─ Result: Toast "Password Verified" → Step 3                │
│                                                                │
│  Step 3: Multi-Factor Authentication (MFA)                    │
│  ├─ Method: Email OTP / TOTP / Both                           │
│  ├─ Input: 6-digit code                                       │
│  └─ Result: Toast "🔓 Login Successful" → Dashboard           │
│                                                                │
└────────────────────────────────────────────────────────────────┘

📊 DATA MODELS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

MEMBER
├─ id: UUID
├─ phoneNumber: string (international format)
├─ fullName: string
├─ pin: string (4 digits)
├─ faydaNumber: string
├─ status: 'active' | 'suspended' | 'inactive'
├─ kycStatus: 'pending' | 'approved' | 'rejected'
└─ registeredAt: timestamp

ADMIN_USER
├─ id: UUID
├─ email: string
├─ passwordHash: string (bcrypt in production)
├─ role: 'super_admin' | 'kyc_approver' | 'dispute_manager' | 'finance_auditor'
├─ permissions: Permission[]
├─ mfaEnabled: boolean
├─ mfaMethod: 'email' | 'totp' | 'both'
├─ status: 'active' | 'inactive' | 'suspended'
└─ createdAt: timestamp

KYC_DOCUMENT
├─ id: UUID
├─ memberId: UUID
├─ documentType: 'id_photo' | 'proof_of_residence' | 'business_license'
├─ status: 'pending' | 'approved' | 'rejected'
├─ submittedAt: timestamp
├─ approvedBy: UUID (admin_user.id)
└─ approvalNotes: string

DISPUTE
├─ id: UUID
├─ complainantId: UUID
├─ respondentId: UUID
├─ description: string
├─ priority: 'low' | 'medium' | 'high' | 'critical'
├─ status: 'open' | 'investigating' | 'resolved' | 'closed'
└─ createdAt: timestamp

FINANCIAL_RECORD
├─ id: UUID
├─ memberId: UUID
├─ type: 'payment' | 'payout' | 'fee' | 'refund'
├─ amount: number
├─ currency: 'ETB' | 'USD'
├─ status: 'completed' | 'pending' | 'failed'
└─ date: timestamp
